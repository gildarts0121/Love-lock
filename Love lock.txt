const boxes = document.querySelectorAll('.box');
const message = document.querySelector('.message');
const correctPassword = ['0','3','1','5'];

// Auto-focus next box
boxes.forEach((box, index) => {
  box.addEventListener('input', () => {
    if(box.value.length === 1 && index < boxes.length - 1) {
      boxes[index+1].focus();
    }
    checkPassword();
  });
});

function checkPassword() {
  const entered = Array.from(boxes).map(b => b.value);
  if(entered.join('') === correctPassword.join('')) {
    // Show animated I Love You
    message.textContent = "I Love You 💖";
    message.classList.add('show');

    // Change lock color
    document.querySelector('.lock').style.background = '#ffb6c1';

    // Create multiple popup I Love You
    for(let i=0;i<20;i++){
      setTimeout(createPopupLove, i*100); // staggered popups
    }
  } else {
    message.classList.remove('show');
    document.querySelector('.lock').style.background = '#ff5e84';
  }
}

function createPopupLove() {
  const popup = document.createElement('div');
  popup.classList.add('popup-love');
  popup.textContent = "I Love You 💖";

  // Random position on screen
  const x = Math.random() * (window.innerWidth - 100);
  const y = Math.random() * (window.innerHeight - 50);
  popup.style.left = x + 'px';
  popup.style.top = y + 'px';

  document.body.appendChild(popup);

  // Remove after animation
  setTimeout(() => {
    popup.remove();
  }, 2000);
}
